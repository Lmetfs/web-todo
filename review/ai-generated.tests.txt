//   1. 有效性：expect(todos[0]).toBeDefined() —— 弱断言，文本存成空串也照样通过
//   2. 有效性：空字符串用例期望值写反（把「没拦住」写成期望 1）
//   3. 污染：todos 声明在文件顶层，被所有用例共享
//   4. 边界：null、超长文本、dataset.id 类型脏值，一条都没有

import { test, expect } from 'vitest';
import { addTodo, toggleTodo, type Todo } from './todo';

// 共享数据（问题 3）
const todos: Todo[] = [];

// 维度2（格式）：正常添加
test('添加一条任务', () => {
  addTodo(todos, '学习TS');
  expect(todos[0]).toBeDefined();
});

// 维度2（格式）：纯空格
test('拒绝纯空格', () => {
  const before = todos.length;
  addTodo(todos, '   ');
  expect(todos.length).toBe(before);
});

// 维度2（格式）：空字符串
test('拒绝空字符串', () => {
  addTodo(todos, '');
  expect(todos.length).toBe(1);
});

// 维度6（重复）：连续添加相同文本
test('连续添加相同文本', () => {
  addTodo(todos, '学习TS');
  addTodo(todos, '学习TS');
  expect(todos.length).toBe(2);
});

// 维度5（状态）：切换完成状态
test('切换完成状态', () => {
  const list: Todo[] = [{ id: 1, text: '学习TS', completed: false }];
  toggleTodo(list, 1);
  expect(list[0].completed).toBe(true);
});
